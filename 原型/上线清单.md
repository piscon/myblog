# 首页：编辑部结构 + 搜索按钮（身份层耦合）

curl -s https://deux.top/en/404.html | grep -c 'There is nothing here'
# 三语言完整性
for l in '' ja/ en/; do curl -s -o /dev/null -w "%{http_code} $l\n" https://deux.top/${l}; done
