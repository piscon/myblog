  /* Row dates inside a year group drop the year prefix: the group heading
     already owns it, so "2026." on every row is zero information. The full
     date stays on the <time datetime> attr and the title tooltip. */
  function timeShort(iso, full) {
    return '<time datetime="' + iso + '" title="' + full + '">' + iso.slice(5).replace('-', '.') + '</time>';
  }

  function yearGroupsHtml(list, opts) {
    opts = opts || {};
    var groups = {};
    list.forEach(function (p) { (groups[p.iso.slice(0, 4)] = groups[p.iso.slice(0, 4)] || []).push(p); });
    /* On a category term page every row carries the same cat as the page
       title — pass it through as a constant column and drop it from rows. */
    if (!opts.hideCat && list.length && list.every(function (p) { return p.cat === list[0].cat; })) opts.hideCat = true;
    return Object.keys(groups).sort(function (a, b) { return b - a; }).map(function (year) {
      var rows = groups[year].map(function (p) {
        return '<a class="post-row' + (opts.hideCat ? ' no-cat' : '') + '" href="' + lp('posts/' + p.slug) + '">' +
          '<span class="p-date">' + timeShort(p.iso, p.date) + '</span>' +
          '<span class="p-title">' + esc(titleOf(p)) + '</span>' +
          (opts.hideCat ? '' : '<span class="p-cat">' + esc(cat(p.cat)) + '</span>') +
          '<span class="p-read">' + t('min_read', { n: p.read }) + '</span>' +
          '</a>';
      }).join('');
      return '<section class="year-group" aria-label="' + year + '">' +
        '<h2 class="year-label">' + year + '<small>' + groups[year].length + '</small></h2>' +
        '<div>' + rows + '</div>' +
        '</section>';
    }).join('');
  }

  function plain(html) {
    var d = document.createElement('div');

  function highlight(text, q) {
    text = String(text || '');
    if (!q) return esc(text);
    var i = text.toLowerCase().indexOf(q.toLowerCase());
    if (i < 0) return esc(text);
    return esc(text.slice(0, i)) + '<mark>' + esc(text.slice(i, i + q.length)) + '</mark>' + esc(text.slice(i + q.length));
  }

  function renderSearch() {
    if (!searchResults) return;
    var q = (searchInput.value || '').trim();
    var ql = q.toLowerCase();
    var list = posts.filter(function (p) {
      if (searchCat !== '全部' && p.cat !== searchCat) return false;
      if (!q) return true;
      var hay = (titleOf(p) + ' ' + descOf(p) + ' ' + p.title + ' ' + p.desc + ' ' + p.tags.join(' ') + ' ' + plain(p.body)).toLowerCase();
      return hay.indexOf(ql) >= 0;
    });

    searchCount.textContent = q ? t('search_count_found', { n: list.length }) : t('search_count_all', { n: list.length });
    document.getElementById('search-total').textContent = t('count_n', { n: list.length });

    if (!list.length) {
      searchResults.innerHTML = '<div class="empty-state">' +
        '<h2>' + esc(t('empty_title', { q: q })) + '</h2>' +
        '<p>' + esc(t('empty_hint')) + '</p>' +
        '<div class="hint-links">' +
        (HINTS[LANG] || HINTS.zh).map(function (w) { return '<button type="button" data-hint="' + esc(w) + '">' + esc(w) + '</button>'; }).join('') +
        '</div></div>';
      syncTagState(q);
      return;
    }

    searchResults.innerHTML = list.map(function (p) {
      return '<a class="result-row" href="' + lp('posts/' + p.slug) + '">' +
        '<span class="res-date">' + timeTag(p.date, p.iso) + '<br>' + esc(cat(p.cat)) + '</span>' +
        '<div><h3>' + highlight(titleOf(p), q) + '</h3><p>' + highlight(descOf(p), q) + '</p></div>' +
        '</a>';
    }).join('');
    syncTagState(q);
  }

  /* ==================================================================
     Article TOC

  var tocIo = null;

  function buildToc(slug) {
    var body = document.getElementById('art-body');
    var inline = document.getElementById('toc-inline');
    var float = document.getElementById('toc-float');
    var count = document.getElementById('toc-count');
    var il = document.getElementById('toc-inline-list');
    var fl = document.getElementById('toc-float-list');
    if (!body || !inline || !float || !il || !fl) return;
    if (tocIo) { tocIo.disconnect(); tocIo = null; }

    var heads = [].slice.call(body.querySelectorAll('h2[id]'));
    if (heads.length < 2) {
      inline.hidden = true; float.hidden = true; float.dataset.enabled = '0';
      il.innerHTML = ''; fl.innerHTML = ''; count.textContent = '';
      return;
    }

    var items = heads.map(function (h) {
      h.setAttribute('tabindex', '-1');
      return '<li><a class="toc-link" href="' + lp('posts/' + slug) + '" data-target="' + h.id + '">' + esc(h.textContent.trim()) + '</a></li>';
    }).join('');
    il.innerHTML = items;
    fl.innerHTML = items;
    count.textContent = t('toc_count', { n: heads.length });
    inline.hidden = false;
    float.hidden = true;
    float.dataset.enabled = '1';

    if ('IntersectionObserver' in window) {
      tocIo = new IntersectionObserver(function (entries) {
        entries.forEach(function (en) { if (en.isIntersecting) setTocActive(en.target.id); });
      }, { rootMargin: '-12% 0px -72% 0px', threshold: 0 });
      heads.forEach(function (h) { tocIo.observe(h); });
    }
    setTocActive(heads[0].id);
  }

  function setTocActive(id) {
    document.querySelectorAll('.toc-link').forEach(function (a) {
      if (a.getAttribute('data-target') === id) a.setAttribute('aria-current', 'location');
      else a.removeAttribute('aria-current');
    });
  }

  function scrollToArticleTarget(id) {
    var el = document.getElementById(id);
    if (!el) return;
    var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    el.scrollIntoView({ behavior: reduce ? 'auto' : 'smooth', block: 'start' });
    el.focus({ preventScroll: true });
  }

  document.addEventListener('click', function (e) {
    var a = e.target.closest('[data-target]');
    if (!a) return;
    e.preventDefault();
    scrollToArticleTarget(a.getAttribute('data-target'));
    var det = a.closest('details');
    if (det) det.open = false;
  });

  function updateTocFloat() {
    var float = document.getElementById('toc-float');
    if (!float) return;
    var view = document.getElementById('view-article');
    if (!view || view.hidden || float.dataset.enabled !== '1' || window.innerWidth < 1200) {
      float.hidden = true;
      return;
    }
    var cover = document.getElementById('art-cover');
    var past = cover && !cover.hidden ? cover.getBoundingClientRect().bottom < 130 : window.scrollY > 260;
    float.hidden = !past;
  }

  /* ==================================================================
     Reading progress
  ================================================================== */
  var readbar = document.getElementById('readbar');
  function updateReadbar() {
    if (!readbar) return;
    var articleView = document.getElementById('view-article');
    if (!articleView || articleView.hidden) {
      readbar.hidden = true;
      updateTocFloat();
      return;
    }
    readbar.hidden = false;
    var doc = document.documentElement;
    var max = doc.scrollHeight - window.innerHeight;
    var pct = max > 0 ? Math.min(1, Math.max(0, window.scrollY / max)) : 0;
    readbar.style.width = (pct * 100).toFixed(2) + '%';
    updateTocFloat();
  }
  window.addEventListener('scroll', updateReadbar, { passive: true });
  window.addEventListener('resize', updateReadbar);

  var lightbox = (function () {
    var el = document.getElementById('lightbox');
    var img = document.getElementById('lb-img');
    var cap = document.getElementById('lb-cap');
    var closeBtn = document.getElementById('lb-close');
    var lastFocus = null;
    if (!el) return { el: null, open: function () {}, close: function () {} };

    function open(node) {
      var fig = node.closest('figure');
      var figcap = fig ? fig.querySelector('figcaption') : null;
      lastFocus = node;
      img.src = node.currentSrc || node.src;
      img.alt = node.alt || '';
      cap.textContent = figcap ? figcap.textContent : '';
      el.hidden = false;
      document.body.style.overflow = 'hidden';
      closeBtn.focus();
    }
    function close() {
      if (el.hidden) return;
      el.hidden = true;
      document.body.style.overflow = '';
      img.src = '';
      if (lastFocus) { lastFocus.focus({ preventScroll: true }); lastFocus = null; }
    }
    closeBtn.addEventListener('click', close);
    el.addEventListener('click', function (e) { if (e.target === el) close(); });
    return { el: el, open: open, close: close };
  })();

  var views = {
    home: document.getElementById('view-home'),
    posts: document.getElementById('view-posts'),
    archives: document.getElementById('view-archives'),
    article: document.getElementById('view-article'),
    search: document.getElementById('view-search'),
    about: document.getElementById('view-about'),
    tax: document.getElementById('view-tax'),
    term: document.getElementById('view-term'),
    guestbook: document.getElementById('view-guestbook'),
    '404': document.getElementById('view-404')
  };

  function parseRoute() {
    var h = location.hash.replace(/^#\/?/, '');
    var qi = h.indexOf('?');
    var query = {};
    if (qi >= 0) {
      var params = new URLSearchParams(h.slice(qi + 1));
      params.forEach(function (v, k) { query[k] = v; });
      h = h.slice(0, qi);
    }
    var parts = h.split('/').filter(Boolean);
    var lang = 'zh';
    if (parts.length && (parts[0] === 'ja' || parts[0] === 'en')) {
      lang = parts.shift();
    }
    var r = { lang: lang, query: query, kind: null, term: null, slug: null };
    currentPath = parts.join('/');
    if (!parts.length) return Object.assign(r, { name: 'home' });
    if (parts[0] === 'posts' && parts[1]) return Object.assign(r, { name: 'article', slug: decodeURIComponent(parts[1]) });
    if (parts[0] === 'posts') return Object.assign(r, { name: 'posts' });
    if (parts[0] === 'archives') return Object.assign(r, { name: 'archives' });
    if (parts[0] === 'search') return Object.assign(r, { name: 'search' });
    if (parts[0] === 'about') return Object.assign(r, { name: 'about' });
    if (parts[0] === 'guestbook') return Object.assign(r, { name: 'guestbook' });
    if (parts[0] === '404') return Object.assign(r, { name: '404' });
    if (parts[0] === 'tags' || parts[0] === 'categories') {
      if (parts[1]) return Object.assign(r, { name: 'term', kind: parts[0], term: decodeURIComponent(parts[1]) });
      return Object.assign(r, { name: 'tax', kind: parts[0] });
    }
    return Object.assign(r, { name: '404' }); /* unknown paths 404, like the real site */
  }

  function showView(name) {
    Object.keys(views).forEach(function (k) {
      if (!views[k]) return;
      views[k].hidden = k !== name;
      views[k].classList.toggle('view-enter', k === name);
    });
  }

  function syncNav(name, kind) {
    var map = { posts: 'posts', article: 'posts', archives: 'posts', tax: kind === 'categories' ? null : 'posts', term: 'posts' };
    document.querySelectorAll('[data-nav]').forEach(function (a) {
      if (map[name] && a.getAttribute('data-nav') === map[name]) a.setAttribute('aria-current', 'page');
      else a.removeAttribute('aria-current');
    });
    var searchLink = document.getElementById('search-link');
    if (searchLink) {
      if (name === 'search') searchLink.setAttribute('aria-current', 'page');
      else searchLink.removeAttribute('aria-current');
    }
  }

  var currentKey = null;

  function route() {
    var r = parseRoute();
    if (r.lang !== LANG) {
      LANG = r.lang;
      applyLangChrome();
      updateInternalLinks();
      renderHomeTexts();
      renderRecent();
      renderLedger();
      renderTagHub();
    } else {
      updateLangLinks();
    }

    showView(r.name);
    syncNav(r.name, r.kind);

    if (r.name === 'home') { document.title = '二重奏'; renderHomeTexts(); }
    if (r.name === 'posts') { document.title = t('nav_posts') + ' · 二重奏'; renderList(); }
    if (r.name === 'archives') { document.title = t('archives_title') + ' · 二重奏'; renderArchives(); }
    if (r.name === 'search') { document.title = t('search_title') + ' · 二重奏'; }
    if (r.name === 'about') { document.title = t('nav_about') + ' · 二重奏'; syncAboutLang(); }
    if (r.name === 'guestbook') {
      document.title = t('guestbook_title') + ' · 二重奏';
      document.getElementById('gb-title').textContent = t('guestbook_title');
      document.getElementById('gb-intro').innerHTML = (I18N[LANG] && I18N[LANG].guestbook) || I18N.zh.guestbook || '';
      document.getElementById('gb-comments').innerHTML = commentsBlock('gb-comments-title');
    }
    if (r.name === 'tax') { document.title = (r.kind === 'tags' ? t('tags_title') : t('cats_title')) + ' · 二重奏'; renderTax(r.kind); }
    if (r.name === 'term') { document.title = (r.kind === 'tags' ? r.term : cat(r.term)) + ' · 二重奏'; renderTerm(r.kind, r.term); }
    if (r.name === '404') {
      document.title = '404 · 二重奏';
      document.getElementById('nf-msg').textContent = t('nf_msg');
      document.getElementById('nf-home-label').textContent = t('nf_home');
      document.getElementById('nf-posts-label').textContent = t('nf_posts');
      document.getElementById('nf-search-label').textContent = t('nf_search');
      document.getElementById('nf-recent-title').textContent = t('nf_recent');
      document.getElementById('nf-recent').innerHTML = posts.slice(0, 3).map(function (x) {
        return '<a class="related-row" href="' + lp('posts/' + x.slug) + '">' +
          '<span class="r-title">' + esc(titleOf(x)) + '</span>' +
          '<span class="r-date">' + timeTag(x.date, x.iso) + '</span></a>';
      }).join('');
    }

    if (r.name === 'article') renderArticle(r.slug);
    if (r.name === 'search') {
      if (typeof r.query.q === 'string' && r.query.q !== searchInput.value) searchInput.value = r.query.q;
      if (r.query.cat && ['全部'].concat(CATS).indexOf(r.query.cat) >= 0) searchCat = r.query.cat;
      else searchCat = '全部';
      var sf = document.getElementById('search-filters');
      if (sf) sf.innerHTML = filterButtons('data-scat', searchCat);
      renderSearch();
    }
    if (r.name === 'posts') {
      listCat = (r.query.cat && ['全部'].concat(CATS).indexOf(r.query.cat) >= 0) ? r.query.cat : '全部';
      renderList();
    }

    var key = r.lang + '/' + r.name + '/' + (r.slug || r.term || r.kind || '');
    if (key !== currentKey) {
      var html = document.documentElement;
      var prevBehavior = html.style.scrollBehavior;
      html.style.scrollBehavior = 'auto';
      window.scrollTo(0, 0);
      html.style.scrollBehavior = prevBehavior;
      var main = document.getElementById('main');
      if (main) main.focus({ preventScroll: true });
    }
    currentKey = key;
    updateReadbar();
    observeReveals();
  }

  window.addEventListener('hashchange', route);

  /* ==================================================================
     Reveal
  ================================================================== */
  var io = null;
  if ('IntersectionObserver' in window && !(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches)) {
    io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        if (en.isIntersecting) { en.target.classList.add('in'); io.unobserve(en.target); }
      });
    }, { rootMargin: '0px 0px -8% 0px', threshold: 0.05 });
  }
  function observeReveals() {
    if (!io) {
      document.querySelectorAll('.reveal').forEach(function (el) { el.classList.add('in'); });
      return;
    }
    document.querySelectorAll('.reveal').forEach(function (el) {
      if (views.home && !views.home.hidden && !el.classList.contains('in')) io.observe(el);
    });
  }

  /* ==================================================================
     Shortcuts: Ctrl/⌘+K and Alt+/ both open search (site parity)
  ================================================================== */
  document.addEventListener('keydown', function (e) {
    var tag = (e.target.tagName || '').toLowerCase();
    var typing = tag === 'input' || tag === 'textarea' || e.target.isContentEditable;

    if ((e.metaKey || e.ctrlKey) && (e.key === 'k' || e.key === 'K')) {
      e.preventDefault();
      if (location.hash !== lp('search')) location.hash = lp('search');
      setTimeout(function () { searchInput.focus(); searchInput.select(); }, 60);
      return;
    }
    if (e.altKey && e.key === '/') {
      e.preventDefault();
      if (location.hash !== lp('search')) location.hash = lp('search');
      setTimeout(function () { searchInput.focus(); }, 60);
      return;
    }
    if (e.key === '/' && !typing) {
      e.preventDefault();
      if (location.hash !== lp('search')) location.hash = lp('search');
      setTimeout(function () { searchInput.focus(); }, 60);
    }
  });

  /* ==================================================================
     Boot
  ================================================================== */

  renderRecent();
  renderLedger();
  renderTagHub();
  route();
})();
